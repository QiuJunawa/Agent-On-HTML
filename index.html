function memTokenize(text) {

  for (let i = 0; i < chars.length - 1; i++) { const b = chars.slice(i, i + 2); tokens[b] = (tokens[b] || 0) + 1; }

}
function memCos(a, b) {
  let dot = 0, na = 0, nb = 0;
  const keys = new Set(Object.keys(a).concat(Object.keys(b)));
  for (const k of keys) { const x = a[k] || 0, y = b[k] || 0; dot += x * y; na += x * x; nb += y * y; }
  if (!na || !nb) return 0;
  return dot / (Math.sqrt(na) * Math.sqrt(nb));
}

    if (!Object.keys(tokens).length) return;
    const tx = db.transaction(MEM_STORE, 'readwrite');
    tx.objectStore(MEM_STORE).add({ text: String(text).slice(0, 1200), tokens, time: Date.now(), source: source || 'chat' });
  } catch (e) {}
}
async function memGetAll() {
  try {
    const db = await memOpen(); if (!db) return [];
    return await new Promise((res) => {
      const req = db.transaction(MEM_STORE, 'readonly').objectStore(MEM_STORE).getAll();
      req.onsuccess = () => res(req.result || []);
      req.onerror = () => res([]);
    });
  } catch (e) { return []; }
}
async function memSearch(query, k) {
  k = k || 4;
  try {
    const q = memTokenize(query);
    if (!Object.keys(q).length) return [];
    const all = await memGetAll();
    return all.map(m => ({ m, score: memCos(q, m.tokens) }))
      .filter(x => x.score > 0.12)
      .sort((a, b) => b.score - a.score)
      .slice(0, k);
  } catch (e) { return []; }
}
async function memCount() { try { return (await memGetAll()).length; } catch (e) { return 0; } }
async function memClear() {
  try {
    const db = await memOpen(); if (!db) return;
    const tx = db.transaction(MEM_STORE, 'readwrite');
    tx.objectStore(MEM_STORE).clear();
  } catch (e) {}
}
async function memTrim() {
  try {
    const all = await memGetAll();
    if (all.length <= MEM_MAX) return;
    all.sort((a, b) => (a.time || 0) - (b.time || 0));
    const drop = all.slice(0, all.length - MEM_MAX).map(m => m.id);
    const db = await memOpen(); if (!db) return;
    const tx = db.transaction(MEM_STORE, 'readwrite');
    const st = tx.objectStore(MEM_STORE);
    drop.forEach(id => st.delete(id));
  } catch (e) {}
}
async function memRemember(userText, aiText) {
  try {
    if (cfg.memoryOn === false) return;
    const q = String(userText || '').trim();
    if (!q) return;
    // 1. 头尾各 40 字，用于 tokenize 和检索（保住开头背景 + 结尾结论）
    const head = q.slice(0, 40).replace(/\s+/g, ' ').trim();
    const tail = q.length > 80 ? q.slice(-40).replace(/\s+/g, ' ').trim() : '';
    const topic = tail ? (head + '…' + tail) : head;
    // 2. 简化版：去掉代码块、压缩空白，取前 80 字，存 text 字段给 UI/扩展用
    const simplified = q
      .replace(/```[\s\S]*?```/g, '[代码]')
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 80);
    // 3. 指针：当前对话文件路径（saveConversation 已执行，sessionFile 有值）
    const pointer = sessionFile || '';
    // 去重
    const hits = await memSearch(topic, 1);
    if (hits.length && hits[0].score > 0.85) return;
    const db = await memOpen();
    if (!db) return;
    const tx = db.transaction(MEM_STORE, 'readwrite');
    tx.objectStore(MEM_STORE).add({
      text: simplified,
      topic: topic,
      tokens: memTokenize(topic),
      pointer: pointer,
      time: Date.now(),
      source: 'chat'
    });
    await memTrim();
  } catch (e) {}
}

let _thumbSignature = '';
let _thumbScheduled = false;
let _thumbLastRun = 0;
// 用户手动滚缩略条时不把 scrollTop 拉回；程序同步触发的 scroll 靠时间戳区分
let _thumbUserScrolling = false;
let _thumbScrollTimer = null;
let _thumbProgrammaticScrollAt = -1e9;
const THUMB_MIN_INTERVAL = 150;
const THUMB_ITEM_H = 8;

function bindThumbScrollGuard(bar) {
  if (bar._thumbScrollBound) return;
  bar._thumbScrollBound = true;
  bar.addEventListener('scroll', () => {
    // scrollTop 被程序赋值也会触发 scroll，150ms 内的视为程序滚动
    if (performance.now() - _thumbProgrammaticScrollAt < 150) return;
    _thumbUserScrolling = true;
    clearTimeout(_thumbScrollTimer);
    _thumbScrollTimer = setTimeout(() => { _thumbUserScrolling = false; }, 500);
  }, { passive: true });
}

function scheduleThumbUpdate() {
  if (_thumbScheduled) return;
  _thumbScheduled = true;
  requestAnimationFrame(() => {
    _thumbScheduled = false;
    const now = performance.now();
    if (now - _thumbLastRun < THUMB_MIN_INTERVAL) {
      setTimeout(scheduleThumbUpdate, THUMB_MIN_INTERVAL - (now - _thumbLastRun));
      return;
    }
    _thumbLastRun = now;
    updateScrollThumbnail();
  });
}

  const hasImages = !!(lastMsg && lastMsg.images && lastMsg.images.length);

  if (cfg.memoryOn !== false) {

    try { await saveConversation(); } catch (e) { console.warn('保存对话失败', e); toast('保存对话失败：' + (e && e.message ? e.message : e), 'err'); } await refreshFS();

        if (lastAssist) await memRemember(lastText, String(lastAssist.content));
      }
    } catch (e) {}
  }
}

// 对话正文预算随模型上下文窗口变化（留 40% 给 system + tools + 输出）
function getChatBudget() {
  const m = (cfg.model || '').toLowerCase();
  let ctx = 128000;  // 默认
  if (m.includes('1m') || m.includes('gemini-1.5') || m.includes('claude-3-5-sonnet') || m.includes('claude-sonnet-4') || m.includes('deepseek-flash') || m.includes('deepseek-v4')) ctx = 200000;
  else if (m.includes('32k')) ctx = 32000;
  else if (m.includes('16k')) ctx = 16000;
  else if (m.includes('8k')) ctx = 8000;
  else if (m.includes('128k') || m.includes('deepseek') || m.includes('glm') || m.includes('qwen') || m.includes('gpt-4o') || m.includes('kimi')) ctx = 128000;
  return Math.floor(ctx * 0.6 * 1.5);  // 中文约 1.5 字符/token（第一批）
}
/* 判断当前模型是否原生支持视觉输入（图片直接塞进 message content 数组） */
function modelSupportsVision() {
  const m = String(cfg.model || '').toLowerCase();
  if (!m) return false;
  const patterns = ['flash', 'vision', '-vl', 'vl-', '4o', '4.1', 'gpt-4', 'claude-3', 'gemini', 'qwen-vl', 'glm-4v', 'yi-vl', 'doubao-1.5-vision', 'doubao-vision'];
  return patterns.some(p => m.includes(p));
}
/* content 可能是字符串（纯文本）或数组（图文混合），统一取纯文本部分用于长度计算/存档/脱敏 */
function contentToText(c) {
  if (Array.isArray(c)) {
    return c.map(part => {
      if (part && part.type === 'text') return String(part.text || '');
      if (part && part.type === 'image_url') return '[图片]';
      return '';
    }).join('\n');
  }
  return String(c || '');
}
function trimConversation(msgs) {
  let total = 0;
  const kept = [];
  for (let i = msgs.length - 1; i >= 0; i--) {
    // 过滤卡片消息（ask/deliver/chart）：它们只是 UI 渲染数据，不是协议消息，
    // 且插在 assistant(tool_calls) 与 tool 之间会破坏 API 要求的消息序列
    const m = msgs[i];
    if (m && (m.ask || m.deliver || m.chart)) continue;
    const len = contentToText(m.content).length + 160;   // 每条消息估算开销
    if (total + len > getChatBudget() && kept.length) break;
    total += len;
    kept.unshift(m);
  }
  const dropped = msgs.length - kept.length;
  if (dropped > 0) {
    kept.unshift({ role: 'system', content: '【较早的消息因过长已被省略。如需早期信息，可用 read_file 查看 /conversations/ 目录下的对话记录。】' });
  }
  // 序列修复：tool 消息必须紧跟在带对应 tool_call_id 的 assistant(tool_calls) 之后；
  // 多工具并行时连续的 tool 响应块只要第一条合法就全部保留
  const fixed = [];
  let inToolBatch = false;
  for (const m of kept) {
    if (m.role === 'tool') {
      if (inToolBatch) { fixed.push(m); continue; }
      const prev = fixed[fixed.length - 1];
      const ok = prev && prev.role === 'assistant' && prev.tool_calls &&
        prev.tool_calls.some(tc => tc.id === m.tool_call_id);
      if (ok) { fixed.push(m); inToolBatch = true; }
      // 孤立的 tool 消息（前面的 assistant(tool_calls) 被裁掉了）：丢弃，避免 API 400
    } else {
      inToolBatch = false;
      fixed.push(m);
    }
  }
  // DeepSeek 等推理模型要求多轮对话时 assistant 消息必须带 reasoning_content 字段（即使为空字符串），
  // 否则报 400 "The reasoning_content in the thinking mode must be passed back to the API"。
  // 老对话/云端恢复的消息可能缺这个字段，这里补全。
  const normalized = fixed.map(m => (m.role === 'assistant' && m.reasoning_content == null) ? { ...m, reasoning_content: '' } : m);
  // 只保留最近 3 轮的 reasoning_content，更早的清空（DeepSeek 多轮对话允许空字符串）
  const KEEP_REASONING_ROUNDS = 3;
  let assistantCount = 0;
  for (let i = normalized.length - 1; i >= 0; i--) {
    if (normalized[i].role === 'assistant') {
      assistantCount++;
      if (assistantCount > KEEP_REASONING_ROUNDS) {
        normalized[i] = { ...normalized[i], reasoning_content: '' };
      }
    }
  }
  return normalized;
}

/* ---------- DeepSeek DSML 工具调用解析（跨 chunk 流式状态机） ----------
   兼容各种写法：单/双竖线（<｜DSML｜ 或 <｜｜DSML｜｜）、标签名 tool_calls/calls、
   闭合斜杠在尖括号后（）或竖线后（<｜｜DSML｜｜ /invoke>）、裸 invoke。 */
function dsmlDecode(s) {
  return String(s || '').replace(/</g, '<').replace(/>/g, '>').replace(/&/g, '&').replace(/"/g, '"').replace(/'/g, "'").replace(/'/g, "'");
}
// DSML 标签通用形态：]*>/gi;
// 解析一段（含完整 invoke 的）DSML 文本，提取标准 tool_calls
function parseDsmlBlock(block) {
  const calls = [];
  let curName = null, curArgs = null, curParam = null, valStart = -1, m;
  DSML_TAG_RE.lastIndex = 0;
  while ((m = DSML_TAG_RE.exec(block))) {
    const tag = m[0];
    // parameter 开标签之后到当前标签之前的文本就是该参数值
    if (curParam != null && valStart >= 0) curArgs[curParam] = (curArgs[curParam] || '') + block.slice(valStart, m.index);
    const close = /^<\s*\//i.test(tag) || /\s*\/(?:tool_calls|calls|invoke|parameter)\s*>\s*$/i.test(tag);
    const nm = (tag.match(/(?:tool_calls|calls|invoke|parameter)/i) || [])[0];
    if (!nm) continue;
    const name = nm.toLowerCase();
    if (close) {
      if (name === 'parameter') { curParam = null; valStart = -1; }
      else if (name === 'invoke' && curName) {
        const args = curArgs || {};
        Object.keys(args).forEach(k => { args[k] = dsmlDecode(String(args[k])); });
        calls.push({ id: 'call_dsml_' + Date.now() + '_' + calls.length, type: 'function', function: { name: curName, arguments: JSON.stringify(args) } });
        curName = null; curArgs = null; curParam = null; valStart = -1;
      }
    } else if (name === 'invoke') {
      const a = tag.match(/name="([^"]*)"/i);
      curName = a ? a[1] : ''; curArgs = {}; curParam = null; valStart = -1;
    } else if (name === 'parameter') {
      const a = tag.match(/name="([^"]*)"/i);
      curParam = a ? a[1] : ''; valStart = DSML_TAG_RE.lastIndex;
    }
  }
  return calls;
}
// 有状态流式过滤器：逐 chunk 喂入，标记一出现即拦截，跨 chunk 拼接完整块后再路由为工具调用
// 代码块（``` 包裹）内的 DSML 标记属于转义范围，原样透传不解析
function createDsmlFilter() {
  const START_RE = /<[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*(?:tool_calls|calls|invoke)\b/i;
  // 只认闭合标签（斜杠在 < 后，或在标签名前），不能把开标签误当闭合
  const closeRe = (invokeOnly) => invokeOnly
    ? /<\/[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*invoke\s*>|<[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*\/invoke\s*>/i
    : /<\/[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*(?:tool_calls|calls)\s*>|<[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*\/(?:tool_calls|calls)\s*>/i;
  // 末尾若可能是某个起始标记的不完整前缀（含标签名写到一半，如 <｜DSML｜tool_c），整段保留
  const CANDS = ['<｜DSML｜tool_calls','<｜DSML｜calls','<｜DSML｜invoke','<｜｜DSML｜｜tool_calls','<｜｜DSML｜｜calls','<｜｜DSML｜｜invoke'].map(s => s.replace(/\s/g, ''));
  const fenceCount = (s) => (s.match(/```+/g) || []).length;
  const tailKeep = (s) => {
    let keep = 0;
    for (let i = 0; i < s.length; i++) {
      if (s[i] === '<') {
        const t = s.slice(i).replace(/\s/g, '');
        if (CANDS.some(c => c.startsWith(t))) keep = s.length - i;
      }
    }
    // 末尾连续反引号可能是不完整的代码块围栏，保留等下一个 chunk
    const bm = s.match(/`+$/);
    if (bm) keep = Math.max(keep, bm[0].length);
    return keep;
  };
  return {
    pending: '', capture: '', capturing: false, startInvoke: false, inCodeBlock: false,
    feed(chunk) {
      const newCalls = [];
      let out = '';
      if (this.capturing) {
        this.capture += chunk;
        const cm = this.capture.match(closeRe(this.startInvoke));
        if (cm) {
          const end = cm.index + cm[0].length;
          const block = this.capture.slice(0, end);
          const rest = this.capture.slice(end);
          this.capture = ''; this.capturing = false; this.startInvoke = false;
          newCalls.push(...parseDsmlBlock(block));
          const sub = this.feed(rest);
          out += sub.text; newCalls.push(...sub.calls);
        }
        return { text: out, calls: newCalls };
      }
      this.pending += chunk;
      // 循环扫描：代码块内的 DSML 标记原样透传，代码块外的触发捕获
      while (true) {
        const sm = this.pending.match(START_RE);
        if (!sm) break;
        // 判断该标记是否在代码块内：已输出状态 + 标记前 pending 中的围栏奇偶
        const before = this.pending.slice(0, sm.index);
        const inCode = this.inCodeBlock ? (fenceCount(before) % 2 === 0) : (fenceCount(before) % 2 === 1);
        if (inCode) {
          // 代码块内：标记当普通文本输出，更新围栏状态，继续扫描后续
          const piece = this.pending.slice(0, sm.index + sm[0].length);
          if (fenceCount(piece) % 2) this.inCodeBlock = !this.inCodeBlock;
          out += piece;
          this.pending = this.pending.slice(sm.index + sm[0].length);
          continue;
        }
        // 代码块外：触发捕获
        const prefix = this.pending.slice(0, sm.index);
        if (fenceCount(prefix) % 2) this.inCodeBlock = !this.inCodeBlock;
        out += prefix;
        this.capture = this.pending.slice(sm.index);
        this.pending = '';
        this.capturing = true;
        this.startInvoke = /invoke\b/i.test(sm[0]);
        const sub = this.feed('');
        return { text: out + sub.text, calls: newCalls.concat(sub.calls) };
      }
      // 无匹配：安全输出（保留末尾不完整前缀）
      const keep = tailKeep(this.pending);
      const toOut = this.pending.slice(0, this.pending.length - keep);
      if (fenceCount(toOut) % 2) this.inCodeBlock = !this.inCodeBlock;
      out += toOut;
      this.pending = this.pending.slice(this.pending.length - keep);
      return { text: out, calls: newCalls };
    },
    flush() {
      const newCalls = [];
      let out = '';
      if (this.capturing) {
        const got = parseDsmlBlock(this.capture);
        if (got.length) {
          newCalls.push(...got);
        } else if (!/<[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*invoke\b/i.test(this.capture)) {
          // fail-open：连 invoke 都没有，说明不是真实工具调用（讨论语法/举例），原样透传
          out += this.capture;
        }
        // 含 invoke 但未闭合：模型中断，丢弃避免乱码
        this.capture = ''; this.capturing = false;
      }
      out += this.pending; this.pending = '';
      return { text: out, calls: newCalls };
    }
  };
}

  const reader = resp.body.getReader();
  const decoder = new TextDecoder('utf-8');
  let buffer = '';
  let gotAnyData = false;
  let finishReason = null;   // length / content_filter / stop / tool_calls
  let sawDone = false;       // 是否收到 [DONE]（流正常收尾标志）

  const READ_IDLE_MS = 90000;   // 90 秒无任何 chunk 判定 idle 超时（避免无限挂起）
  while (true) {
    let readResult;
    try {
      readResult = await Promise.race([
        reader.read(),
        new Promise((_, reject) => setTimeout(() => reject(new Error('__IDLE_TIMEOUT__')), READ_IDLE_MS))
      ]);
    } catch (e) {
      if (e && e.message === '__IDLE_TIMEOUT__') {
        text += '\n\n[⚠ 流读取 idle 超时（' + Math.round(READ_IDLE_MS / 1000) + ' 秒无数据）。可能是模型思考过久被中间层断开，或网络代理超时。已收到的内容保留]';
        break;
      }
      throw e;
    }
    const { done, value } = readResult;
    if (done) break;
    gotAnyData = true;
    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split('\n');
    buffer = lines.pop();
    for (const rawLine of lines) {
      const line = rawLine.trim();
      if (!line || !line.startsWith('data:')) continue;
      const data = line.slice(5).trim();
      if (data === '[DONE]') { sawDone = true; continue; }
      let json; try { json = JSON.parse(data); } catch (e) { continue; }
      if (json.error) { msgEl.remove(); throw new Error('API 错误：' + (json.error.message || JSON.stringify(json.error))); }
      if (json.usage) usage = json.usage;   // 流式最后一条消息携带实际 token 用量
      const choice = json.choices && json.choices[0];
      if (!choice) continue;
      if (choice.finish_reason) finishReason = choice.finish_reason;   // 记录停止原因（length/content_filter/stop/tool_calls）
      const delta = choice.delta || {};
      // DeepSeek 等推理模型：先发 reasoning_content（思考中），正文稍后才到
      if (delta.reasoning_content) { thinking += delta.reasoning_content; scheduleRender(); }
      if (delta.content) {
        dsmlIngest(delta.content);
      }
      if (delta.tool_calls) {
        for (const tc of delta.tool_calls) {
          const idx = tc.index != null ? tc.index : 0;
          if (!toolAcc[idx]) toolAcc[idx] = { id: '', type: 'function', function: { name: '', arguments: '' } };
          if (tc.id) toolAcc[idx].id = tc.id;
          if (tc.function) {
            if (tc.function.name) toolAcc[idx].function.name += tc.function.name;
            if (tc.function.arguments) toolAcc[idx].function.arguments += tc.function.arguments;
          }
        }
      }
    }
  }

  // 如果没收到任何 SSE 数据但 buffer 里有残留，尝试当整段 JSON 解析
  if (!gotAnyData && buffer.trim()) {
    try {
      const j = JSON.parse(buffer.trim());
      const choice = j.choices && j.choices[0];
      if (choice && choice.message) {
        if (choice.message.reasoning_content) thinking = choice.message.reasoning_content;
        if (choice.message.content) {
          const r = dsmlFilter.feed(choice.message.content);
          if (r.text) text += r.text;
          r.calls.forEach(tc => { toolAcc[toolAcc.length] = tc; });
        }
        if (choice.message.tool_calls) {
          choice.message.tool_calls.forEach((tc, i) => {
            toolAcc[i] = { id: tc.id || '', type: 'function', function: { name: (tc.function && tc.function.name) || '', arguments: (tc.function && tc.function.arguments) || '{}' } };
          });
        }
      }
      if (thinking || text) renderBubble();
    } catch (e) { /* 忽略，走下面的空响应判断 */ }
  }

  // 收尾：flush 状态机（未闭合块内若有完整 invoke 仍采纳，半个前缀标记不渲染）
  const flush = dsmlFilter.flush();
  if (flush.text) text += flush.text;
  flush.calls.forEach(tc => { toolAcc[toolAcc.length] = tc; });
  // 流层/模型层停止诊断：截断、内容过滤、流提前关闭（都要显式告诉用户，不要静默停止）
  let stopDiag = '';
  if (finishReason === 'length') {
    stopDiag += '\n\n[⚠ 输出达到 max_tokens 上限（' + payload.max_tokens + '）被截断。可在设置中调大「输出 token 上限」，或让 AI 分多次输出]';
  } else if (finishReason === 'content_filter') {
    stopDiag += '\n\n[⚠ 输出被内容过滤中断]';
  } else if (!sawDone && gotAnyData) {
    stopDiag += '\n\n[⚠ 流被提前关闭（未收到 [DONE]），可能是网络代理或服务端 idle timeout。已收到的内容保留]';
  }
  if (!text.trim() && stopDiag) text = stopDiag.trim();

  const hasToolCalls = toolAcc.filter(Boolean).length > 0;
  const hasThinking = !!(thinking && thinking.trim());
  if (!text.trim() && !hasToolCalls && !hasThinking) {
    msgEl.remove();
    throw new Error('API 返回了空响应（无内容、无工具调用、无思考过程）。可能是：模型名称错误、Base URL 不正确、或 API 不支持 stream 模式。');
  }
  if (!text.trim() && hasToolCalls && !hasThinking) {
    // 只有工具调用没有文本和思考，移除空气泡（工具卡片会单独显示）
    msgEl.remove();
  }
  if (!text.trim() && !hasToolCalls && hasThinking) {
    // 模型只输出了思考、没生成正文：把思考摘要回填到 content，避免"静默停止"
    const brief = thinking.replace(/\s+/g, ' ').trim().slice(0, 300);
    text = '⚠ 模型只输出了思考过程，没有生成正式回复。可能原因：max_tokens 截断、内容过滤、或模型认为任务已完成。\n\n思考摘要：' + brief + '\n\n[请点「重新生成」重试，或在设置中调大「输出 token 上限」]';
  }
  if (stopDiag && text.indexOf(stopDiag.trim()) < 0) text += stopDiag;
  const toolCalls = toolAcc.filter(Boolean).map((tc, i) => ({ id: tc.id || ('call_' + Date.now() + '_' + i), type: 'function', function: { name: tc.function.name, arguments: tc.function.arguments || '{}' } }));
  // 清理残留的 DSML 标记（未闭合/孤立标签），避免渲染出乱码
  text = text.replace(/<\/?[｜]{1,2}\s*DSML\s*[｜]{1,2}\s*\/?(?:tool_calls|calls|invoke|parameter|tool_calls|calls|invoke|parameter)[^>]*>/gi, '').replace(/<[｜]{1,2}\s*DSML\s*[｜]{0,2}[^>\n]*$/gi, '').trim();
  // 兜底：GLM4.7 等模型输出的  文本格式，从显示中剔除并解析成工具调用
  if (text.includes('')) {
    const parsed = parseTextToolCalls(text);
    if (parsed.length > 0) {
      parsed.forEach(tc => toolCalls.push(tc));
      text = text.replace(/[\s\S]*?<\/tool_call>/g, '').trim();
    }
  }
  if (thinking || text) renderBubble();   // 结束前确保最终内容已渲染（含被 rAF 合并的尾部）
  scrollBottom();
  return { content: text, toolCalls, usage, thinking };
}

      return;
    }
    // 流式解析 SSE，累积 content
    const reader = resp.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '', title = '', thinking = '';
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';
      for (const line of lines) {
        const s = line.trim();
        if (!s.startsWith('data:')) continue;

        if (data === '[DONE]') continue;
        try {
          const j = JSON.parse(data);
          const delta = j.choices && j.choices[0] && j.choices[0].delta;
          if (delta) {
            if (delta.content) title += delta.content;
            if (delta.reasoning_content) thinking += delta.reasoning_content;
          }
        } catch (e) {}
      }
    }
    // content 为空时从思考过程里找最可能的标题：清洗过程性语句，避免把思考过程当标题
    if (!title.trim() && thinking.trim()) {

      const lines = thinking.split(/\n+/)
        .map(l => l.replace(/^[#*\-\d\.\s>]+/, '').replace(/[。．.！!？?]+$/, '').trim())

      for (let i = lines.length - 1; i >= 0; i--) {

      }
    }
    title = title.trim().replace(/^["'「『]+|["'」』]+$/g, '').slice(0, 30);
    console.log('[generateTitle] 收到标题:', JSON.stringify(title));
    if (title) {

      if (titleEl) titleEl.textContent = title;
      const s = sessions.find(x => x.id === currentSessionId);
      if (s) { s.title = title; s.updatedAt = Date.now(); saveSessions(); }
    } else {

    }
  } catch (e) {

    else toast('标题生成失败：' + (e && e.message ? e.message : e), 'err');
  }
}

/* ---------- 文本工具调用兜底解析（GLM4.7 等不支持 OpenAI tool_calls 的模型） ---------- */
function parseTextToolCalls(content) {
  const calls = [];
  const re = /([\s\S]*?)<\/tool_call>/g;
  let m;
  while ((m = re.exec(content)) !== null) {
    const body = m[1].trim();
    const nameMatch = body.match(/^([a-zA-Z_][a-zA-Z0-9_]*)/);
    if (!nameMatch) continue;
    const name = nameMatch[1];
    const args = {};
    const argRe = /([\s\S]*?)<\/arg_key>\s*([\s\S]*?)<\/arg_value>/g;
    let am;
    while ((am = argRe.exec(body)) !== null) {
      args[am[1].trim()] = am[2].trim();
    }
    calls.push({
      id: 'txt_' + Date.now() + '_' + calls.length,
      type: 'function',
      function: { name, arguments: JSON.stringify(args) }
    });
  }
  return calls;
}

/* ---------- 工具调度器：资源感知并行 + 失败补偿重试 ---------- */
const WRITE_OPS = new Set(['write_file', 'append_file', 'delete_path', 'move_path', 'create_dir', 'kb_write', 'kb_delete', 'kb_commit']);

// 从工具参数提取操作的资源标识，同资源的操作必须串行
function extractResource(name, args) {
  if (name.startsWith('todo_')) return 'todo:__all__';
  // 目录级操作（删除/移动）用全树锁，避免与子文件操作竞态
  if (name === 'delete_path' || name === 'move_path') return 'vfs:__tree__';
  if (['write_file', 'append_file', 'read_file', 'create_dir'].includes(name)) return 'vfs:' + (args.path || '');
  if (name === 'kb_commit') return 'kb:__commit__';
  if (['kb_write', 'kb_delete', 'kb_read'].includes(name)) return 'kb:' + (args.path || '');
  if (['fetch_url', 'web_fetch', 'download_url'].includes(name)) return 'url:' + (args.url || '');
  if (name === 'web_search') return 'search:' + (args.query || '');
  return 'indep:' + name + ':' + Math.random().toString(36).slice(2, 8);
}

// 判断结果是否失败（需要重试）——只对瞬时/可重试错误重试，不对 4xx 客户端错误重试
function isToolFailure(result) {
  const s = String(result || '');
  // 明确的客户端错误不重试（Key错/权限不足/文件不存在/参数错误）
  if (s.includes('HTTP 401') || s.includes('HTTP 403') || s.includes('HTTP 404') || s.includes('HTTP 400')) return false;
  if (s.includes('已禁用') || s.includes('未配置') || s.includes('需要在设置') || s.includes('暂存区为空')) return false;
  // 可重试：限流、服务端错误、网络超时、ref冲突、临时失败
  if (s.includes('HTTP 429') || s.includes('HTTP 5') || s.includes('超时') || s.includes('timeout') || s.includes('Failed to fetch') || s.includes('NetworkError')) return true;
  if (s.includes('冲突') || s.includes('conflict') || s.includes('ref') || s.includes('sha')) return true;
  if (s.startsWith('工具执行失败')) return true;
  return false;
}

// 带指数退避重试的执行
async function execWithRetry(tc, execOne, maxRetries) {
  let lastResult = null;
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const r = await execOne(tc);
    lastResult = r;
    if (!isToolFailure(r.result)) return r;
    if (attempt < maxRetries) {
      const delay = 400 * Math.pow(2, attempt);  // 400ms, 800ms, 1600ms
      await new Promise(res => setTimeout(res, delay));
    }
  }
  return lastResult;
}

// 资源感知调度：同资源串行，不同资源并行；kb_commit 最后执行
async function scheduleToolCalls(toolCalls, execOne, maxRetries) {
  // 分离 kb_commit（必须最后执行）
  const commitCalls = toolCalls.filter(tc => tc.function.name === 'kb_commit');
  const normalCalls = toolCalls.filter(tc => tc.function.name !== 'kb_commit');

  // 按资源分组
  const groups = new Map();
  for (const tc of normalCalls) {
    let args = {};
    try { args = JSON.parse(tc.function.arguments || '{}'); } catch (e) {}
    const res = extractResource(tc.function.name, args);
    if (!groups.has(res)) groups.set(res, []);
    groups.get(res).push(tc);
  }

  // 全树锁：有 vfs:__tree__ 组时，它和所有 vfs:* 组必须串行（tree 先执行），非 vfs 组可并行
  const hasTreeLock = groups.has('vfs:__tree__');
  const vfsGroups = [];  // 需串行的 vfs 组（含 tree）
  const otherGroups = []; // 可并行的非 vfs 组
  for (const [key, calls] of groups) {
    if (hasTreeLock && key.startsWith('vfs:')) vfsGroups.push({ key, calls });
    else otherGroups.push({ key, calls });
  }
  // vfs 组按 tree 优先排序
  vfsGroups.sort((a, b) => a.key === 'vfs:__tree__' ? -1 : b.key === 'vfs:__tree__' ? 1 : 0);

  // 执行：vfs 串行组 + 非 vfs 并行组同时启动
  const allPromises = [];
  if (vfsGroups.length > 0) {
    allPromises.push((async () => {
      const results = [];
      for (const g of vfsGroups) {
        for (const tc of g.calls) results.push(await execWithRetry(tc, execOne, maxRetries));
      }
      return results;
    })());
  }
  for (const g of otherGroups) {
    allPromises.push((async () => {
      const results = [];
      for (const tc of g.calls) results.push(await execWithRetry(tc, execOne, maxRetries));
      return results;
    })());
  }
  const normalResults = (await Promise.all(allPromises)).flat();

  // kb_commit 最后执行（在所有 kb 操作完成后）
  let commitResults = [];
  if (commitCalls.length > 0) {
    for (const tc of commitCalls) commitResults.push(await execWithRetry(tc, execOne, maxRetries));
  }

  // 按原始顺序排列结果
  const allResults = [...normalResults, ...commitResults];
  return toolCalls.map(tc => allResults.find(r => r.tc === tc));
}

/* ---------- 预算控制 ---------- */
function checkBudget(budget) {
  const reasons = [];
  if (cfg.maxToolCalls > 0 && budget.toolCalls >= cfg.maxToolCalls) reasons.push('工具调用次数达上限 ' + cfg.maxToolCalls);
  if (cfg.maxWallTime > 0 && (Date.now() - budget.startTime) / 1000 >= cfg.maxWallTime) reasons.push('执行时间达上限 ' + cfg.maxWallTime + 's');
  if (cfg.maxInputTokens > 0 && budget.inputTokens >= cfg.maxInputTokens) reasons.push('输入 token 达上限 ' + cfg.maxInputTokens);
  if (cfg.maxOutputTokens > 0 && budget.outputTokens >= cfg.maxOutputTokens) reasons.push('输出 token 达上限 ' + cfg.maxOutputTokens);
  if (cfg.maxCost > 0 && budget.cost >= cfg.maxCost) reasons.push('成本达上限 $' + cfg.maxCost);
  return reasons;
}

// 按模型关键词匹配大致价格（美元/百万 token），用于预算估算
function getModelPricing(model) {
  const m = (model || '').toLowerCase();
  // 免费/极便宜模型
  if (m.includes('glm-4.7-flash') || m.includes('glm-4-flash') || m.includes('glm-4.5-air')) return { in: 0, out: 0 };
  if (m.includes('glm-4-air')) return { in: 0.001, out: 0.001 };
  if (m.includes('glm') || m.includes('chatglm')) return { in: 0.6, out: 2.2 };
  // DeepSeek V4 系列（人民币定价换算美元，汇率约7.2，取高峰时段+缓存未命中的保守估算）
  // flash:  输入缓存未命中 2元/M → $0.28,  输出 8元/M → $1.11
  // v4-pro: 输入缓存未命中 9元/M → $1.25,  输出 27元/M → $3.75
  // （缓存命中价格低一个量级，空闲时段约为高峰的一半，预算估算统一取最坏情况）
  if (m.includes('deepseek') && m.includes('v4-pro')) return { in: 1.25, out: 3.75 };
  if (m.includes('deepseek') && m.includes('reasoner')) return { in: 0.28, out: 2.22 };
  if (m.includes('deepseek') && (m.includes('chat') || m.includes('v3'))) return { in: 0.14, out: 0.56 };
  if (m.includes('deepseek')) return { in: 0.28, out: 1.11 };  // flash (默认)
  // GPT 系列（从便宜到贵排序）
  if (m.includes('gpt-4o-mini') || m.includes('gpt-3.5')) return { in: 0.15, out: 0.60 };
  if (m.includes('gpt-4o')) return { in: 2.5, out: 10.0 };
  if (m.includes('gpt-4-turbo') || m.includes('gpt-4') || m.includes('o1')) return { in: 10.0, out: 30.0 };
  // Claude（Haiku 便宜，其他贵）
  if (m.includes('haiku')) return { in: 0.25, out: 1.25 };
  if (m.includes('claude')) return { in: 3.0, out: 15.0 };
  // Qwen
  if (m.includes('qwen-turbo')) return { in: 0.15, out: 0.45 };
  if (m.includes('qwen-plus')) return { in: 0.8, out: 2.0 };
  if (m.includes('qwen-max')) return { in: 2.4, out: 7.2 };
  if (m.includes('qwen')) return { in: 0.5, out: 2.0 };
  // Kimi
  if (m.includes('kimi') || m.includes('moonshot')) return { in: 0.6, out: 2.4 };
  return { in: 0.5, out: 1.5 };  // 默认估算
}

function estimateCost(inputTokens, outputTokens) {
  const price = getModelPricing(cfg.model);
  return (inputTokens / 1e6) * price.in + (outputTokens / 1e6) * price.out;
}

async function agentLoop() {
  let loops = 0;
  const maxLoops = Math.max(1, Number(cfg.maxLoops) || 20);
  // 预算追踪（调度层硬控，不靠模型自觉）
  const budget = {
    startTime: Date.now(),
    toolCalls: 0,
    inputTokens: 0,
    outputTokens: 0,
    cost: 0
  };
  while (loops < maxLoops) {
    if (chatAbort && chatAbort.signal.aborted) return;
    // 预算检查：发起新请求前先查
    const over = checkBudget(budget);
    if (over.length > 0) {
      addErrorMsg('⛔ 调度层预算超限，已停止：' + over.join('；') + '。已完成 ' + loops + ' 轮、' + budget.toolCalls + ' 次工具调用。');
      return;
    }
    loops++;
    let { content, toolCalls, usage, thinking } = await streamChat();
    if (usage) {
      lastUsage = usage;
      budget.inputTokens += (usage.prompt_tokens || usage.input_tokens || 0);
      budget.outputTokens += (usage.completion_tokens || usage.output_tokens || 0);
      budget.cost += estimateCost(usage.prompt_tokens || usage.input_tokens || 0, usage.completion_tokens || usage.output_tokens || 0);
    }
    // 兜底：部分模型（如 GLM4.7）不支持 OpenAI tool_calls 字段，会把工具调用以  文本形式输出
    if (toolCalls.length === 0 && content && content.includes('')) {
      const parsed = parseTextToolCalls(content);
      if (parsed.length > 0) {
        toolCalls = parsed;
        content = content.replace(/[\s\S]*?<\/tool_call>/g, '').trim();
      }
    }
    if (toolCalls.length > 0) {
      conversation.push({ role: 'assistant', content: content || null, tool_calls: toolCalls, reasoning_content: thinking || '' });
      // 预算检查：工具调用数
      budget.toolCalls += toolCalls.length;
      const over2 = checkBudget(budget);
      if (over2.length > 0) {
        for (const tc of toolCalls) {
          conversation.push({ role: 'tool', tool_call_id: tc.id, content: '预算超限，未执行' });
        }
        addErrorMsg('⛔ 调度层预算超限，已停止：' + over2.join('；') + '。本批 ' + toolCalls.length + ' 次工具调用未执行。');
        return;
      }
      const execOne = async (tc) => {
        if (chatAbort && chatAbort.signal.aborted) return { tc, result: '已停止' };
        let args = {}; try { args = JSON.parse(tc.function.arguments || '{}'); } catch (e) {}
        const card = addToolCard(tc.function.name, args);
        let livePre = null;
        try {
          const result = await executeTool(tc.function.name, args, (chunk) => {
            if (!livePre) {
              livePre = document.createElement('pre');
              livePre.className = 'route-live';
              livePre.style.cssText = 'margin:8px 0 0;padding:8px;background:#0f1115;border-radius:6px;font-size:12px;white-space:pre-wrap;word-break:break-word;max-height:300px;overflow-y:auto;color:#c8cdd6';
              card.querySelector('.tool-result-wrap').appendChild(livePre);
              card.querySelector('.tool-status').textContent = '路由中…';
            }
            livePre.textContent = chunk;
            scrollBottom();
          });
          finishToolCard(card, true, result);
          return { tc, result };
        }
        catch (e) {
          const result = '工具执行失败：' + e.message;
          finishToolCard(card, false, result);
          return { tc, result };
        }
      };
      // 资源感知调度：同资源串行，不同资源并行；kb_commit 最后执行；失败自动重试（最多2次）
      const results = await scheduleToolCalls(toolCalls, execOne, 2);
      const resultMap = new Map(results.filter(Boolean).map(r => [r.tc.id, r]));
      for (const tc of toolCalls) {
        const r = resultMap.get(tc.id);
        conversation.push({ role: 'tool', tool_call_id: tc.id, content: String(r ? r.result : '已停止').slice(0, 30000) });
      }
      continue;
    }
    conversation.push({ role: 'assistant', content: content || '', reasoning_content: thinking || '' });
    break;
  }
  if (loops >= maxLoops) addErrorMsg(`⚠️ 已达到最大工具循环次数（${maxLoops}），任务可能未完成。可在设置中调大该值。`);
  // 结尾兜底：只有思考没有正文的 assistant 会让用户以为"莫名其妙停了"，直接把诊断贴到对话里
  const lastMsg = conversation[conversation.length - 1];
  if (lastMsg && lastMsg.role === 'assistant' && !String(lastMsg.content || '').trim() && String(lastMsg.reasoning_content || '').trim()) {
    addErrorMsg('⚠ 本轮异常结束：模型只输出了思考过程，没有生成正文。已执行工具 ' + ((typeof budget !== 'undefined' && budget && budget.toolCalls) || 0) + ' 次。请点「重新生成」重试，或在设置中调大「输出 token 上限」。');
  }
}

function convToMarkdown() {
  const lines = [];
  for (const m of conversation) {
    if (m.role === 'user') {
      lines.push('## 用户\n\n' + contentToText(m.content));
    } else if (m.role === 'assistant') {
      if (m.reasoning_content && m.reasoning_content.trim()) lines.push('## 思考过程\n\n' + contentToText(m.reasoning_content));
      if (m.content) lines.push('## 助手\n\n' + contentToText(m.content));
      if (m.tool_calls) {
        for (const tc of m.tool_calls) {
          lines.push('## 调用工具: ' + (tc.function && tc.function.name || '') + '\n\n```json\n' + ((tc.function && tc.function.arguments) || '{}') + '\n```');
        }
      }
    } else if (m.role === 'tool') {
      lines.push('## 工具结果\n\n```\n' + contentToText(m.content) + '\n```');
    }
  }
  return lines.join('\n\n');
}

async function collectSensitiveSecrets() {
  const secrets = [];
  const add = v => { if (v && typeof v === 'string' && v.length >= 4 && !secrets.includes(v)) secrets.push(v); };
  // ai-profiles.json 里每组的 apiKey
  try {
    const raw = await VFS.getContent('/config/ai-profiles.json');
    if (raw) {
      const j = JSON.parse(raw);
      const list = Array.isArray(j) ? j : (j.profiles || []);
      for (const p of list) if (p && p.apiKey) add(p.apiKey);
    }
  } catch (e) { /* 文件不存在或解析失败则跳过 */ }
  // 当前设置里保存的密钥（AI 可能通过工具调用参数或返回内容带出）
  add(cfg.apiKey); add(cfg.giteeToken); add(cfg.wdPass); add(cfg.gistToken);
  // 长值优先替换，避免短值先被替换导致长值替换不完整
  return secrets.sort((a, b) => b.length - a.length);
}

function redactText(text, secrets) {
  if (!secrets.length) return text;
  let out = text;
  for (const s of secrets) {
    if (out.includes(s)) out = out.split(s).join('****');
  }
  return out;
}

function extFromMime(mime) {
  const m = String(mime || '').toLowerCase();
  if (m.includes('png')) return '.png';
  if (m.includes('jpeg') || m.includes('jpg')) return '.jpg';
  if (m.includes('gif')) return '.gif';
  if (m.includes('webp')) return '.webp';
  if (m.includes('bmp')) return '.bmp';
  if (m.includes('svg')) return '.svg';
  return '.png';
}

async function handlePasteImage(e) {
  const items = e.clipboardData && e.clipboardData.items;
  if (!items) return;
  for (const item of items) {
    if (item.type && item.type.startsWith('image/')) {
      e.preventDefault();
      const file = item.getAsFile();
      if (!file) continue;
      if (file.size > UPLOAD_MAX_BYTES) { toast('粘贴图片超过 5MB，已跳过', 'err'); continue; }
      try {
        await requireVfs();
        const dataUrl = await readFileAsDataURL(file);
        const ts = new Date();
        const pad = (n) => String(n).padStart(2, '0');
        const name = 'paste-' + ts.getFullYear() + pad(ts.getMonth()+1) + pad(ts.getDate()) + '-' + pad(ts.getHours()) + pad(ts.getMinutes()) + pad(ts.getSeconds()) + '-' + Math.floor(Math.random()*1000) + extFromMime(file.type);

  }

  }

async function evictLocalContent() {
  // 保留 meta（文件树），清空所有文件正文与内存缓存；之后按需从云端拉取
  VFS.cache.clear();
  const textPaths = Object.keys(VFS.meta).filter(k => VFS.meta[k].type === 'file');
  if (memoryOnly) {
    for (const p of textPaths) delete memStore.content[p];
  } else {
    const db = await getDB();
    for (const p of textPaths) {
      await new Promise((resolve, reject) => {
        const tx = db.transaction('content', 'readwrite');
        tx.objectStore('content').delete(p);
        tx.oncomplete = () => resolve();
        tx.onerror = () => reject(tx.error);
      });
    }
  }
  toast('已上传并清空文本文件正文，图片保留本地，文件将按需从云端拉取');
  await refreshFS();
}

/* 粗略估算 token 数：中文约 1.5 字符/token，英文约 4 字符/token，取混合估算 */
function estimateTokens(text) {
  if (!text) return 0;
  const s = String(text);
  let cjk = 0, other = 0;
  for (const ch of s) { if (/[\u4e00-\u9fff\u3000-\u303f\uff00-\uffef]/.test(ch)) cjk++; else other++; }
  return Math.ceil(cjk / 1.5 + other / 4);
}
function updateTokenBar() {
  const inp = $('input'); const inputText = inp ? inp.value : '';
  const ti = $('tokenInput'); const tc = $('tokenContext');
  if (ti) ti.textContent = '输入 ' + estimateTokens(inputText) + ' tokens';
  if (tc) {
    if (lastUsage) {
      const hit = lastUsage.prompt_cache_hit_tokens || 0;
      const miss = lastUsage.prompt_cache_miss_tokens || (lastUsage.prompt_tokens ? lastUsage.prompt_tokens - hit : 0);
      const total = hit + miss;
      const rate = total ? (hit / total * 100).toFixed(1) : '0';
      const out = lastUsage.completion_tokens || 0;
      tc.textContent = `缓存 ${rate}% | 命中 ${hit} 未命中 ${miss} | 输出 ${out}`;
    } else {
      const ctxText = conversation.map(m => contentToText(m.content)).join('\n');
      tc.textContent = '对话 ~' + estimateTokens(ctxText) + ' tokens';
    }
  }
}

      ed.style.display = '';
    } else {
      // 切到预览
      btn.dataset.mode = 'preview';
      btn.textContent = '源码';
      ed.style.display = 'none';
      pane.style.display = '';
      const path = currentFile || '';
      const mdPreview = $('mdPreview');
      const htmlPreview = $('htmlPreview');
      if (mdPreview) mdPreview.style.display = /\.md$/i.test(path) ? '' : 'none';
      if (htmlPreview) { htmlPreview.style.display = /\.html?$/i.test(path) ? '' : 'none'; if (/\.html?$/i.test(path)) htmlPreview.srcdoc = ed.value; }
      if (mdPreview && /\.md$/i.test(path)) { mdPreview.innerHTML = mdToHtml(ed.value); renderMathIn(mdPreview); }
    }
  });
  const editor = $('editor');
  if (editor) {
    editor.addEventListener('keydown', (e) => {
      if ((e.ctrlKey || e.metaKey) && e.key === 's') { e.preventDefault(); saveEditor(); }
      if (e.key === 'Tab') { e.preventDefault(); const ta = e.target; const s = ta.selectionStart, en = ta.selectionEnd; ta.value = ta.value.slice(0, s) + '  ' + ta.value.slice(en); ta.selectionStart = ta.selectionEnd = s + 2; }
    });
  }
  on('btnDelFile', 'click', async () => {
    if (!currentFile) { toast('没有打开的文件'); return; }
    const path = currentFile;
    // 两段式确认：第一击进入确认态，再点一次才删除（不依赖 confirm，iframe/触摸环境可用）
    const btn = $('btnDelFile');
    if (btn.dataset.armed !== path) {
      btn.dataset.armed = path;
      btn.textContent = '确认删除?';
      toast('再点一次「确认删除?」删除 ' + path + '（不可恢复）', 'warn');
      setTimeout(() => { if (btn.dataset.armed === path) { delete btn.dataset.armed; btn.textContent = '删除'; } }, 5000);
      return;
    }
    delete btn.dataset.armed; btn.textContent = '删除';
    try { await requireVfs(); await VFS.deletePath(path); closeEditor(); await refreshFS(); toast('已删除'); }
    catch (err) { toast('删除失败：' + (err && err.message ? err.message : err), 'err'); }
  });
  on('btnCloudUp', 'click', async (e) => { e.preventDefault(); await cloudUpload(); });
  on('btnCloudDown', 'click', async (e) => { e.preventDefault(); await cloudDownload(); });
  const beSel = $('cfgCloudBackend');
  if (beSel) beSel.addEventListener('change', updateCloudFields);
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeSettings(); });
}
